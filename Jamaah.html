-- ============================================================ -- Profil jamaah lengkap + foto (privat) -- Jalankan sekali di Supabase > SQL Editor, SEBELUM memakai index.html baru. -- Asumsi: biro.id bisa dibandingkan sebagai teks dengan nama folder pertama -- pada path foto (aplikasi menyimpan di "/.jpg"). -- ============================================================ alter table public.jamaah add column if not exists foto text, -- path di bucket privat 'jamaah-foto' add column if not exists jk text, -- 'L' / 'P' add column if not exists tgl_lahir date, add column if not exists nik text, add column if not exists alamat text, add column if not exists paspor text, add column if not exists darurat text, -- nama & no. HP kontak darurat add column if not exists catatan text; -- Bucket PRIVAT: foto jamaah adalah data pribadi, tidak boleh punya tautan publik. -- Aplikasi menampilkannya lewat tautan sementara (signed URL, 10 menit). insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types) values ('jamaah-foto', 'jamaah-foto', false, 1048576, array['image/jpeg']) on conflict (id) do update set public = false, file_size_limit = 1048576, allowed_mime_types = array['image/jpeg']; -- Admin biro hanya boleh mengakses folder biro-nya sendiri. -- Subquery ke tabel biro tunduk pada RLS biro, jadi hanya biro milik pengguna yang terlihat. drop policy if exists "jamaah_foto_biro" on storage.objects; create policy "jamaah_foto_biro" on storage.objects for all to authenticated using ( bucket_id = 'jamaah-foto' and exists (select 1 from public.biro b where b.id::text = (storage.foldername(name))[1]) ) with check ( bucket_id = 'jamaah-foto' and exists (select 1 from public.biro b where b.id::text = (storage.foldername(name))[1]) );
